我们要做到：

- 输入1个或多个辅导员名称，输出TA（们）负责的寝室的门牌号
- 寝室类型匹配（识别四人寝/六人寝）
- 插入相框和姓名
- 确保寝室信息栏（辅导员、专业那两栏）对齐
- 输出时不同辅导员负责的寝室分开
- 同一个园区的门牌放在同一个docx文档里，每一页是一个新的门牌
- 处理多辅导员/专业（混寝）情况
- 处理未住满寝室情况（如6人寝但只有4个）
- 提供友好的交互界面
- 提供帮助文档

我的计划使用Python实现，用pandas读取并处理表格数据，docx库生成门牌文档，docx2pdf格式转换，customtkinter实现交互界面，pyinstaller打包成可执行文件  
整个程序分为两个模块：process_data.py（负责读取、处理数据并生成门牌文档）和GUI.py（用户界面展示）

### 数据处理与门牌生成

#### 表格数据读取/处理

问题：如何将同一个寝室的人归类到一起？  
思路：观察表格数据可知，同一个寝室的人员是相邻的，那么我们可以直接从上向下一行一行地遍历表格，并根据“宿舍”列和“宿舍楼”列将相同寝室的信息装入同一个列表中。  
问题：如何判断一个寝室由哪个辅导员负责？  
思路：为方便起见，同时也为了避免重复生成，每个寝室根据第一个学生的辅导员判断该寝室属于哪个辅导员  
问题：如何生成类似“龙A 633”这样的寝室门牌？  
思路：每行“宿舍楼”列第一个字符+正则搜索“宿舍楼”列的字母+“宿舍”列

```python
import os
import re
import pandas as pd
en_pattern = re.compile("[a-zA-Z]+")
def read_xlsx(teacher:str, path:str):
    """
    要提取的列：
    B列  姓名
    D列  院系
    E列  专业全称
    G列  年级
    H列  辅导员
    I列  宿舍楼
    K列  宿舍
    M列  床位数
    :param teacher: 辅导员名字，用于过滤数据
    :return: 该辅导员负责的学生的宿舍信息     [[楼名, 宿舍, 院系, 年级, 床位数, [辅导员], [专业(全称)], [名字1, 名字2,...]], .....]
    """
    dormitory_index = {}    # 存储每个寝室的索引，这样就可以把时间复杂度减少到O(1)
    neglect = set()         # 如果第一个人的辅导员不符但第二个人符合，那么剩下的还是会被记录，所以用neglect来存储第一个人的辅导员不符的寝室
    dormitory_information = []
    try:
        column_index = get_column_index(path)
        df = pd.read_excel(path, usecols=column_index)
        for index, row in df.iterrows():
            name, academy, major, grade, teach, dormitory, num, bed = row
            # 转换门牌号
            dormitory = dormitory[0] + en_pattern.findall(dormitory)[0] if en_pattern.findall(dormitory) else dormitory[0]
            num = f'{dormitory} {num}'
            if (teach != teacher and dormitory_index.get(num) is None) or num in neglect:
                neglect.add(num)
                continue


            # 如果有索引代表这是同一个寝室的，只把名字添加进去，没有说明正在处理一个新寝室添加索引并添加完整的信息
            if dormitory_index.get(num) is None:
                if int(bed) > 4:
                    bed = 6
                else:
                    bed = 4
                # 包装
                info = [dormitory, num, academy, grade, bed, [teach], [major], [name]]  # 辅导员和专业可能不止一个
                dormitory_index[num] = len(dormitory_information)
                dormitory_information.append(info)
            else:
                info = dormitory_information[dormitory_index.get(num)]
                info[-1].append(name)
                if teach not in info[-3]:
                    info[-3].append(teach)
                if major not in info[-2]:
                    info[-2].append(major)

        for dormitory in dormitory_information:
            # 最后检查一遍列表，填补空床
            while len(dormitory[-1]) < dormitory[4]:
                dormitory[-1].append(None)

    except PermissionError:
        messagebox.showerror("错误", "文件被占用，请关闭WPS表格/Excel后重试")
        return []
    except KeyError:
        messagebox.showerror("错误", "表格格式不正确，请按照使用说明调整表格格式")
        return []
    except Exception as e:
        messagebox.showerror("错误", f"读取文件时出错：{e}")
        return []
    return dormitory_information
```

通过上面的代码，我们就能得到dormitory_information列表，里面每一个元素都是一个独立完整的寝室信息

然后处理寝室信息栏，这是最难的，因为需要对齐  
我们可以把这两行信息放入一个2行4列的无框表格中，这样可以确保大部分情况能够对齐，但由于如下的问题，仍有一部分寝室难以对齐：

- 有些寝室有2个甚至3个辅导员/专业
- 辅导员辅导员的名字长度有2个字的、3个字的甚至还有4个字的
- 每个专业名称长度不同，有些专业名称里还带有数字甚至括号

这就很难办，所以我只能写一堆if判断尽可能地确保大部分可以对齐

```python
def set_academy(doc, major: list, grade, academy, teacher: list):
    """
    使用表格设置基本信息
    :param doc: 要编辑的word文档
    :param major: 专业
    :param grade: 年级
    :param academy: 学院
    :param teacher: 辅导员
    """

    min_space = 10

    major = '、'.join(major)
    teacher = '、'.join(teacher)
    kh = major.count('(')
    am_len = max(len(academy), len(major)) - kh * 3
    cur_len = max(len(academy), len(major))
    major_academy_fill = am_len + 2
    if "(3+2)" in major and major.count("、") == 0:
        major = major.center(10, "_")

    if major_academy_fill > min_space:
        if len(major) < len(academy):
            major = major.center(major_academy_fill, "_")
        elif len(major) > len(academy):
            academy = academy.center(major_academy_fill, "_")
    else:
        major = major.center(min_space, '_')
        academy = academy.center(min_space, '_')

    grade = f"{grade}级"
    if len(teacher) > 3:
        grade = grade.center(len(teacher) + 3, "_")

    if len(teacher) == 2:   # 我觉得地球上应该不会存在名字只有一个字的人
        teacher += '_'

    if cur_len >= 18 or am_len > 16:
        pt = 14
    elif am_len <= 10:
        pt = Info_Pt         # 小二
    # elif am_len > 16:
    #     pt = 14             # 四号
    else:
        pt = 16              # 三号

    table = doc.add_table(rows=2, cols=4)
    table.autofit = False
    table.alignment = WD_PARAGRAPH_ALIGNMENT.CENTER

    # 第一行
    cell_academy_str = table.cell(0, 0)
    cell_academy_str.text = '二级学院'
    set_cell_font(cell_academy_str, Font, Pt(pt), True)  # 设置字体为宋体，字号16，加粗
    disable_cell_wrap(cell_academy_str)

    cell_academy = table.cell(0, 1)
    cell_academy.text = academy
    set_cell_font(cell_academy, Font, Pt(pt), True, True)
    disable_cell_wrap(cell_academy)

    cell_teacher_str = table.cell(0, 2)
    cell_teacher_str.text = '辅导员'
    set_cell_font(cell_teacher_str, Font, Pt(pt), True)
    disable_cell_wrap(cell_teacher_str)

    cell_teacher = table.cell(0, 3)
    cell_teacher.text = teacher
    set_cell_font(cell_teacher, Font, Pt(pt), True, True)
    disable_cell_wrap(cell_teacher)

    # 第二行
    cell_major_str = table.cell(1, 0)
    cell_major_str.text = '专    业'
    set_cell_font(cell_major_str, Font, Pt(pt), True)
    disable_cell_wrap(cell_major_str)

    cell_major = table.cell(1, 1)
    cell_major.text = major
    set_cell_font(cell_major, Font, Pt(pt), True, True)
    disable_cell_wrap(cell_major)

    cell_grade_str = table.cell(1, 2)
    cell_grade_str.text = '年  级'
    set_cell_font(cell_grade_str, Font, Pt(pt), True)
    disable_cell_wrap(cell_grade_str)

    cell_grade = table.cell(1, 3)
    cell_grade.text = grade
    set_cell_font(cell_grade, Font, Pt(pt), True, True)
    disable_cell_wrap(cell_grade)

    # 设置宽度
    # 一般情况
    if pt == 18:
        cell_academy_str.width = cell_major_str.width = Cm(3)
        cell_teacher_str.width = cell_grade_str.width = Cm(2.5)
        if '(' in major:
            cell_academy.width = cell_major.width = Cm(7.25)
        else:
            cell_academy.width = cell_major.width = Cm(6.91)
        cell_teacher.width = cell_grade.width = Cm(2.5)

    # 混寝情况
    elif len(major) > 20:
        cell_academy_str.width = cell_major_str.width = Cm(2.5)
        cell_teacher_str.width = cell_grade_str.width = Cm(2)
        cell_academy.width = cell_major.width = Cm(12.67)
        cell_teacher.width = cell_grade.width = Cm(2)

    else:
        cell_academy_str.width = cell_major_str.width = Cm(2.76)
        if am_len <= 14 and '(' not in major:   ########################################### <=13
            cell_academy.width = cell_major.width = Cm(8.52)
        else:
            cell_academy.width = cell_major.width = Cm(10)
        if len(teacher) >= 3 and len(major) <= 20:
            if pt < 18:
                cell_teacher.width = cell_grade.width = Cm(4.4)
                cell_teacher_str.width = cell_grade_str.width = Cm(2.3)
            else:
                cell_teacher.width = cell_grade.width = Cm(5)
        else:
            cell_teacher_str.width = cell_grade_str.width = Cm(2.21)
            cell_teacher.width = cell_grade = Cm(2.44)

    if teacher.count('、') == 1 and pt == Info_Pt:
        cell_teacher.width = cell_grade.width = Cm(5.22)
    elif teacher.count('、') >= 2:
        cell_teacher.width = cell_grade.width = Cm(5.4)

    if cur_len >= 14 and pt <= 16 and len(teacher) <= 3:
        cell_teacher.width = cell_grade.width = Cm(2.44)

    p = doc.add_paragraph()
    p.paragraph_format.line_spacing = 0.5
```

```python
def main(teacher, excel_path):

    dormitory_information: list = read_xlsx(teacher, excel_path)
    
    total_dorms = len(dormitory_information)
    processed_count = 0
    
    for location, num, academy, grade, bed, teacher, major, name in dormitory_information:
        try:
            if '(' in academy:
                academy = academy[:academy.index('(')]  # 过滤括号内的内容
            elif '（' in academy:
                academy = academy[:academy.index('（')]

            path = f"OUTPUT/{academy}/{teacher[0]}"
            if not os.path.exists(path):
                os.makedirs(path)

            if os.path.exists(os.path.join(path, f'{location}.docx')):
                document = Document(os.path.join(path, f'{location}.docx'))
                document.add_page_break()
            else:
                document = Document()
            set_paper_size(document)
            set_dormitory_num(document, num)

            set_academy(document, major, grade, academy, teacher)
            if len(name) == 4:
                name1, name2, name3, name4 = name
                set_four_info(document, name1, name2, name3, name4)
            else:
                name1, name2, name3, name4, name5, name6 = name
                set_six_info(document, name1, name2, name3, name4, name5, name6)

            document.save(os.path.join(path, f"{location}.docx"))

        except Exception as e:
            messagebox.showerror("错误", f"{teacher} {num} 处理失败：\n{e}")
            print(f'[!] {num} 处理失败：', e)
```

- 检查系统是否支持pdf转换
- 允许用户选择是否转换为pdf

界面部分都放在GUI.py里，用的是customtkinter——比原生tkinter好看一些，也自带深色/浅色外观，省得自己配一整套颜色。窗口整体就是一个从上往下的grid：第0行是作者信息，第1行是文件选择，第2、3行是“选择教师”和搜索框，第4行是教师列表，第5行是pdf复选框，第6行是生成按钮，第7行是帮助入口，第8行是进度条。

#### 窗口尺寸

问题：程序拷到别人电脑上就显示不正常，尤其是系统缩放比设成150%、200%的时候。  
思路：先告诉Windows“这个程序自己会处理DPI”，然后所有尺寸都按100%缩放为基准来设计，运行时再按系统缩放因子放大。

```python
# 设置 DPI 感知（Windows）
try:
    ctypes.windll.shcore.SetProcessDpiAwareness(2)  # Per-Monitor DPI Aware
except Exception:
    pass

# 基础设计尺寸（以 100% 缩放为基准）
BASE_WIDTH = 450
BASE_HEIGHT = 680
MIN_WIDTH = 380
MIN_HEIGHT = 550
```

```python
# 获取系统缩放因子
self.scale_factor = self._get_scaling() if hasattr(self, '_get_scaling') else 1.0

# 计算窗口尺寸（基于基础尺寸）
self.screen_width = int(BASE_WIDTH * self.scale_factor)
self.screen_height = int(BASE_HEIGHT * self.scale_factor)
self.geometry(f"{self.screen_width}x{self.screen_height}")

# 设置最小尺寸，防止窗口过小导致部件显示不全
self.minsize(int(MIN_WIDTH * self.scale_factor), int(MIN_HEIGHT * self.scale_factor))
self.resizable(False, False)
```

窗口本身不允许手动缩放，以免用户把窗口拖小之后列表和按钮互相挤。但界面内部还是配了grid权重，让教师列表那一行（row=4）吃掉所有剩余高度：

```python
# 配置主窗口的 grid 权重，实现响应式布局
self.grid_columnconfigure(0, weight=1)
# 滑动区行（row=4）设置权重，允许垂直方向扩展
self.grid_rowconfigure(4, weight=1)
```

这样不管字体、缩放比怎么变，上面的固定行都保持原样，中间那块列表区自己伸缩。

顺带一提，窗口失焦时背景色会从`#EEF4F9`变成`#F3F3F3`，用于模仿Windows原生程序效果。

#### 文件选择

问题：选完文件要马上把表格里的辅导员列出来，但表格动辄几千行，直接在主线程里读，界面就会白屏假死。  
思路：文件对话框只负责拿到路径，读取交给后台线程，读完再用`after`把刷新界面的事送回来。

```python
def select_file(self):
    """选择文件并读取教师信息"""
    file_path = filedialog.askopenfilename(
        title="请选择表格文件",
        filetypes=[("表格文件", "*.xlsx *.xls *.csv")]
    )
    if file_path:
        self.selected_file_path = file_path
        self.file_label_var.set("文件名：" + os.path.basename(self.selected_file_path))

        thread = threading.Thread(target=self.create_teacher_list_thread, daemon=True)
        thread.start()

def create_teacher_list_thread(self):
    """ 在另一个线程中读取教师信息，防止窗口卡住 """
    try:
        self.teacher_dict = get_teacher_dict(self.selected_file_path)
    except PermissionError:
        messagebox.showerror("错误", "文件被占用，请关闭WPS表格/Excel后重试")
    except KeyError:
        messagebox.showerror("错误", "表格格式不正确，请按照使用说明调整表格格式")
    except Exception as e:
        messagebox.showerror("错误", f"读取文件时出错：\n{e}")
    self.after(0, self.update_teacher_list)
```

这里的原则是：tkinter不是线程安全的，所以子线程里只碰数据，所有界面操作一律通过`self.after(0, ...)`排队回主线程执行。`get_teacher_dict`返回的是`{辅导员: 学院}`字典，键取自表格的“辅导员”列、值取自“院系”列（顺手把学院名后面括号里的内容切掉），后面的列表就是照着它画的。

#### 教师列表与全选

问题：怎么把辅导员一行一行排开，右边还要显示学院名，并且支持全选？列表还可能会很长，得有滚动条。  
思路：用一个`CTkScrollableFrame`当列表容器，第0行放全选、之后一个辅导员一行，行内用grid分两列，把列权重交给学院名那一列。

```python
def add_teacher_row(self, teacher_name, academy_name):
    row_frame = ctk.CTkFrame(
        self.scrollable_frame,
        fg_color="transparent",
        bg_color="transparent"
    )
    row_frame.grid(row=self.teacher_cnt, sticky="ew")

    # 配置列权重，确保学院名称标签能正确对齐
    row_frame.grid_columnconfigure(0, weight=1)
    row_frame.grid_columnconfigure(1, weight=0)

    check_box = ctk.CTkCheckBox(
        row_frame,
        text=teacher_name,
        width=10, font=("微软雅黑", 16),
        checkbox_width=18,
        checkbox_height=18,
        border_width=2,
        corner_radius=0,
        hover_color="skyblue",
        fg_color="#1E90FF",
        command=lambda n=teacher_name: self.select_teacher(n, check_box.get())
    )
    check_box.grid(row=0, column=0, sticky="w", padx=(5, 150), pady=5)

    academy_label = ctk.CTkLabel(
        row_frame,
        text=academy_name,
        font=("微软雅黑", 16),
        anchor="e"
    )
    academy_label.grid(row=0, column=1, sticky="ew")

    # 存储行引用，用于搜索过滤
    self.teacher_rows[teacher_name] = {
        "frame": row_frame,
        "checkbox": check_box,
        "academy": academy_name,
        "visible": True
    }

    self.teacher_cnt += 1
```

学院名放在第1列、`anchor="e"`右对齐，这样一来几十行排下来，学院名会整齐地贴在右边，看着很清楚。

全选没有另外写一套逻辑，它就是滚动区里的一行，只不过行号固定是0，而且永远在列表清空之后第一个被装上。勾选时遍历`teacher_rows`，把当前可见的行全部选上：

```python
def all_select(self, is_selected):
    """全选/取消全选，只影响当前显示的行"""
    if is_selected:
        # 只选择当前显示的教师
        for name, row_data in self.teacher_rows.items():
            if row_data["visible"]:
                self.selected_teachers.add(name)
                if not row_data["checkbox"].get():
                    row_data["checkbox"].select()
    else:
        # 只取消当前显示的教师
        for name, row_data in self.teacher_rows.items():
            if row_data["visible"]:
                self.selected_teachers.discard(name)
                if row_data["checkbox"].get():
                    row_data["checkbox"].deselect()
```

每次重新选文件都要先把旧列表清掉、索引归零，不然新表格的人会接在旧数据下面：

```python
def update_teacher_list(self):
    """更新教师列表"""
    # 先清空滚动区
    for widget in self.scrollable_frame.winfo_children():
        widget.destroy()

    # 行索引归零，清空教师行引用
    self.teacher_cnt = 1
    self.teacher_rows = {}
    # 将教师信息填入滚动区
    for teacher, academy in self.teacher_dict.items():
        self.add_teacher_row(teacher, academy)
    # 清空搜索框
    self.search_entry.delete(0, "end")
```

另外滚动区的canvas里还垫了一张半透明的学院logo当背景。做法是把logo缩放到合适大小、按alpha通道调透明度，然后用`create_image`画在canvas中央，再用`tag_lower`压到最底层，这样翻列表的时候不会盖住文字：

```python
def center_logo(self, event=None):
    """将logo绘制在滚动区中央"""
    if hasattr(self, 'logo_photo'):
        # 获取canvas可见区域的中心
        canvas_width = self.inner_canvas.winfo_width()
        canvas_height = self.inner_canvas.winfo_height()

        center_x = canvas_width // 2
        center_y = canvas_height // 2

        # 删除旧的logo
        self.inner_canvas.delete("background_logo")

        # 在中央绘制logo（使用最低层级）
        self.inner_canvas.create_image(
            center_x, center_y,
            image=self.logo_photo,
            anchor="center",
            tags="background_logo"
        )

        # 确保logo在最底层
        self.inner_canvas.tag_lower("background_logo")
```

#### 搜索

问题：辅导员一多，在一长串名字里找人就很难受了。  
思路：监听搜索框的按键，每次输入就遍历一遍教师行，匹配的显示、不匹配的藏起来，教师名和学院名都参与匹配。

```python
def on_search_changed(self, event=None):
    """搜索框内容改变时触发过滤"""
    search_text = self.search_entry.get().strip().lower()
    self.filter_teachers(search_text)

def filter_teachers(self, search_text):
    """根据搜索词过滤教师列表"""
    for name, row_data in self.teacher_rows.items():
        if not search_text or search_text in name.lower() or search_text in row_data["academy"].lower():
            row_data["frame"].grid()
            row_data["visible"] = True
        else:
            row_data["frame"].grid_remove()
            row_data["visible"] = False
```

- 隐藏用的是`grid_remove`而不是`destroy`，行还在，清空搜索就能直接恢复，不用重新建一遍界面；
- `visible`这个标记除了过滤要用，全选也要看它，所以“先搜‘张’再点全选”只会选中搜索结果里的老师，很符合直觉；
- 搜索框的宽度是按滚动区的宽度减掉清除按钮算出来的，这样搜索框和列表的左右边界是对齐的。

```python
# 固定尺寸常量（先定义，供后续组件使用）
self.teacher_box_width = 320
self.teacher_box_height = 300

# 清除按钮宽度（固定像素）
clear_btn_width = 60
# 搜索框宽度（固定像素，与滚动区宽度协调）
search_entry_width = int(self.teacher_box_width - clear_btn_width - 10)
```

清除按钮其实就是把输入框清空，再手动调一次过滤：

```python
def clear_search(self):

```

#### 检查pdf转换

问题：pdf转换用的是docx2pdf，底层调的是Microsoft Word的转换引擎。如果用户电脑上没装Word，或者WPS抢了注册表，用户勾上“生成后转换为PDF”，就会在转换开始时报错  
思路：程序一启动就先生成一个测试文档并转换，成功才允许勾选。

```python
def check_pdf_conversion():
    """ 检查系统是否支持PDF转换 """
    # 先创建一个测试docx文件
    doc = Document()
    doc.add_paragraph("helloworld")
    doc.save('test.docx')
    try:
        convert('test.docx', 'test.pdf')
        try:
            os.remove('test.docx')
            os.remove('test.pdf')
        except:
            pass
        return True
    except Exception as e:
        messagebox.showerror("Error", str(e))
        try:
            os.remove('test.docx')
        except:
            pass
        return False
```

这个检查本身也要几秒钟，所以同样丢到后台线程里去，如果不能转换则弹出警告框提醒用户：

```python
def pdf_convert_checker(self):
    self.pdf_check_flag = check_pdf_conversion()
    if not self.pdf_check_flag:
        self.convert_pdf_flag = False
        self.pdf_checkbox.deselect()
        messagebox.showwarning("警告", "您的系统不支持PDF转换功能！\n如生成的文件布局不正确，请使用程序目录中的 调整布局.bas 来调整，具体使用方法参见 使用说明.docx")
```

这里有个小坑：检查是异步的，用户完全可能在检查结束前就点了复选框，这时候`pdf_check_flag`还不存在。所以勾选的处理用`try/except AttributeError`兜底，真点了就暂时当作可以转换：

```python
def change_pdf_checkbox_state(self):
    try:
        if self.pdf_check_flag:
            self.convert_pdf_flag = not self.convert_pdf_flag
        else:
            messagebox.showwarning("警告", "您的系统不支持PDF转换功能！\n如生成的文件布局不正确，请使用程序目录中的 调整布局.bas 来调整，具体使用方法参见 使用说明.docx")
            self.convert_pdf_flag = False
            self.pdf_checkbox.deselect()
    except AttributeError:
            self.convert_pdf_flag = not self.convert_pdf_flag
```

#### 生成进度条

问题：用户可能需要生成上百个寝室，docx生成加pdf转换要几十秒，这段时间界面完全没反应，用户很容易以为程序卡死然后强行关掉；而且连点几下“开始生成”还会重复生成。  
思路：一个辅导员开一个线程去生成，进度通过回调汇总；任务期间按钮直接禁用并改字。

```python
def generate_threads(self):
    """ 每个老师一个线程 """
    if not self.selected_file_path:
        messagebox.showinfo("提示", "请选择表格文件")
        return
    if not self.selected_teachers:
        messagebox.showinfo("提示", "您还没有选择任何教师")
        return

    # 转换时开始按钮不可点击
    self.generate_button.configure(state="disabled", text="生成中")

    # 显示进度条并初始化
    self.progress_bar.grid()
    self.progress_bar.set(0)

    # 使用计数器来跟踪完成的线程
    self.completed_tasks = 0
    self.total_tasks = len(self.selected_teachers)

    # 用于跟踪进度
    self.progress_tracker = {}
    for teacher in self.selected_teachers:
        self.progress_tracker[teacher] = 0
    threading.Thread(target=self.create_threads, daemon=True).start()
```

进度条显示的是所有辅导员进度的平均值，所以每个辅导员一个进度槽，谁转完了就把自己的槽填满：

```python
def generate_doormats(self, teacher):
    def progress_callback(current, total):
        # 更新特定教师的进度
        self.progress_tracker[teacher] = current / total
        # 计算总体进度
        avg_progress = sum(self.progress_tracker.values()) / len(self.progress_tracker)
        # 在主线程中更新进度条
        self.after(0, lambda: self.progress_bar.set(avg_progress))

    main(teacher, excel_path=self.selected_file_path, progress_callback=progress_callback)
```

主线程这边用一个计数器盯着所有线程有没有跑完，跑完之后再看要不要接着转pdf：

```python
def increment_completed_count(self):
    """增加完成计数并在完成后更新UI"""
    self.completed_tasks += 1
    if self.completed_tasks >= self.total_tasks:

        # 重置进度条
        self.progress_bar.set(0)
        if self.convert_pdf_flag:
            # 转换时按钮显示为“转换中”
            self.generate_button.configure(state="disabled", text="转换中")
            # 将PDF转换操作放到后台线程执行，避免阻塞GUI
            threading.Thread(target=self.run_pdf_conversion, daemon=True).start()
        else:
            # 完成后开始按钮可点击，移除进度条
            self.progress_bar.grid_remove()
            self.generate_button.configure(state="normal", text="开始生成")
            os.startfile(os.path.join(os.getcwd(), 'OUTPUT'))
            messagebox.showinfo(f"提示", f"生成完成！\n位置：{os.path.join(os.getcwd(), 'OUTPUT')}")
```

pdf转换同样带进度：`docx_to_pdf`先数一遍要转的docx有多少个，每转完一个就回调一次，所以进度条会从0重新涨到100%。转完之后同样用`after`回主线程收尾——恢复按钮、打开输出目录、弹提示：

```python
def run_pdf_conversion(self):
    """在后台线程中执行PDF转换"""
    try:
        docx_to_pdf(teachers=self.selected_teachers, progress_callback=self.update_pdf_progress)
    except Exception as e:
        self.after(0, lambda: messagebox.showerror("错误", f"PDF转换出错：\n{e}"))
    finally:
        # 转换完成后在主线程中更新UI
        self.after(0, self.on_pdf_conversion_complete)
```

进度条没开始时是隐藏的，但它外面的`progress_frame`高度写死、并且关掉了`grid_propagate`，这样进度条出现和消失的时候，界面不会跟着上下跳一下：

```python
self.progress_frame = ctk.CTkFrame(self, fg_color="transparent", height=35)
self.progress_frame.grid(row=8, column=0, padx=20, pady=(0, 10), sticky="ew")
self.progress_frame.grid_propagate(False)  # 禁止子组件改变Frame大小
```

- 使用宏，这也是目前在没有Microsoft Word的情况下唯一的办法
